import React, { useState } from "react";
import { View, Text, TextInput, TouchableOpacity, FlatList, StyleSheet } from "react-native";
import { Video } from "expo-av";

export default function App() {
  const [loggedIn, setLoggedIn] = useState(false);
  const [username, setUsername] = useState("");
  const [voucher, setVoucher] = useState("");
  const [stream, setStream] = useState(null);

  const channels = [
    { name: "القناة 1", url: "https://test-streams.mux.dev/x36xhzz/x36xhzz.m3u8" },
    { name: "القناة 2", url: "https://test-streams.mux.dev/test_001/stream.m3u8" },
    { name: "القناة 3", url: "https://test-streams.mux.dev/dai-discontinuity-dash/manifest.mpd" }
  ];

  const login = () => {
    if (username.trim() && voucher.trim()) {
      setLoggedIn(true);
    }
  };

  // 🔐 شاشة الدخول
  if (!loggedIn) {
    return (
      <View style={styles.loginContainer}>
        <Text style={styles.title}>📡 شبكة النادر نت</Text>

        <TextInput
          placeholder="اسم المستخدم"
          placeholderTextColor="#aaa"
          onChangeText={setUsername}
          style={styles.input}
        />

        <TextInput
          placeholder="كرت الدخول"
          placeholderTextColor="#aaa"
          onChangeText={setVoucher}
          style={styles.input}
        />

        <TouchableOpacity onPress={login} style={styles.loginBtn}>
          <Text style={styles.loginText}>دخول</Text>
        </TouchableOpacity>
      </View>
    );
  }

  // 📺 المشغل
  if (stream) {
    return (
      <View style={styles.videoContainer}>
        <TouchableOpacity onPress={() => setStream(null)} style={styles.backBtn}>
          <Text style={{ color: "white" }}>⬅ رجوع</Text>
        </TouchableOpacity>

        <Video
          source={{ uri: stream }}
          style={styles.video}
          useNativeControls
          resizeMode="contain"
          shouldPlay
        />
      </View>
    );
  }

  // 📺 القنوات
  return (
    <View style={styles.container}>
      <Text style={styles.header}>📺 القنوات المباشرة</Text>

      <FlatList
        data={channels}
        keyExtractor={(item) => item.name}
        contentContainerStyle={{ paddingBottom: 20 }}
        renderItem={({ item }) => (
          <TouchableOpacity
            onPress={() => setStream(item.url)}
            style={styles.card}
          >
            <Text style={styles.cardText}>{item.name}</Text>
            <Text style={styles.play}>▶ تشغيل</Text>
          </TouchableOpacity>
        )}
      />
    </View>
  );
}
